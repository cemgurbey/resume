% Full-width section bar, overhanging the text block on both sides.
\newcommand{\sectionband}[1]{%
  \par\vspace{\abovesection}%
  \makebox[\textwidth][l]{\hspace*{-\bandpad}%
    \bandbox{\textwidth+2\bandpad}{\bandh}{%
      \headfont\bfseries\sectionsize\color{accent}\MakeUppercase{#1}}}%
  \par\vspace{\belowsection}}

% NB: every \entrybold stays inside its own { } group. Left unbraced it leaks
% past the \par and turns the bullet list underneath bold too.
\newcommand{\subentry}[2]{%
  \entryfont\subentrysize
  {\entrybold\color{ink}\MakeUppercase{#1}}%
  \hfill{\entrybold\color{ink}#2}\par}
